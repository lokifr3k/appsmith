WITH updated AS (
    UPDATE work_orders 
    SET 
        target_qty = target_qty + {{ Math.ceil(tbl_production_plan.triggeredRow.safe_to_push * 1.08) }},
        fabric_meters = ROUND(CAST(fabric_meters AS numeric) + CAST({{ FabricCalculator.calculateMeters(get_order_for_push.data[0]?.item_size, get_order_for_push.data[0]?.color, tbl_production_plan.triggeredRow.safe_to_push, 0) }} AS numeric), 2)
    WHERE order_no = '{{tbl_production_plan.triggeredRow.order_no}}'
      AND status = 'Not Started'
    RETURNING *
)
INSERT INTO work_orders (order_no, item_details, target_qty, fabric_meters)
SELECT 
    '{{tbl_production_plan.triggeredRow.order_no}}', 
    '{{tbl_production_plan.triggeredRow.item_details}}', 
    {{ Math.ceil(tbl_production_plan.triggeredRow.safe_to_push * 1.08) }},
    {{ FabricCalculator.calculateMeters(get_order_for_push.data[0]?.item_size, get_order_for_push.data[0]?.color, tbl_production_plan.triggeredRow.safe_to_push, 0) }}
FROM (SELECT 1) t
WHERE NOT EXISTS (SELECT * FROM updated);
