UPDATE export_orders 
SET 
    sent_qty = COALESCE(sent_qty, 0) + {{this.params.qty}},
    invoice_details = CONCAT_WS(', ', invoice_details, 'Inv: ' || '{{inp_invoice_no.text}}' || ' (' || '{{inp_invoice_date.formattedDate}}' || ') - Qty: ' || {{this.params.qty}})
WHERE order_no = '{{this.params.order_no}}';
