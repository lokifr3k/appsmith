WITH FactoryMath AS (
    SELECT 
        si.order_no,
        eo.item_size,
        eo.color,
        eo.folding_type,
        eo.logo,
        si.cartons,
        si.target_qty AS required_pcs,
        
        -- 1. Available Printed: Total Printed MINUS what already left on a truck
        GREATEST(0, COALESCE(eo.produced_qty, 0) - COALESCE(eo.sent_qty, 0)) AS available_printed,
        
        -- 2. Available Folded: (Ordered - Balance = Total Folded) MINUS what already left on a truck
        GREATEST(0, (COALESCE(eo.ordered_qty, 0) - COALESCE(eo.balance_qty, 0)) - COALESCE(eo.sent_qty, 0)) AS available_folded
        
    FROM shipment_items si
    LEFT JOIN export_orders eo ON si.order_no = eo.order_no
    WHERE si.shipment_id = {{tbl_shipments.selectedRow.id || 0}}
)
SELECT 
    order_no,
    item_size,
    color,
    folding_type,
    logo,
    cartons,
    required_pcs,
    
    -- FACTORY TO-DO LIST (Comparing Required against Net Available Stock)
    GREATEST(0, required_pcs - available_printed) AS qty_to_print,
    GREATEST(0, required_pcs - available_folded) AS qty_to_fold,

    -- DYNAMIC FACTORY STATUS
    CASE 
        WHEN GREATEST(0, required_pcs - available_folded) = 0 THEN '🟢 Ready to Load'
        WHEN GREATEST(0, required_pcs - available_printed) = 0 THEN '🟡 Needs Folding'
        ELSE '🔴 Needs Printing'
    END AS action_status

FROM FactoryMath;
