WITH RawData AS (
    SELECT 
        si.order_no,
        eo.item_size,
        eo.color,
        eo.folding_type,
        eo.logo,
        si.cartons,
        si.target_qty,
        COALESCE(eo.stock_qty, 0) AS stock,
        COALESCE(eo.sent_qty, 0) AS sent,
        COALESCE(eo.produced_qty, 0) AS produced,
        COALESCE(eo.folded_qty, 0) AS folded 
    FROM shipment_items si
    LEFT JOIN export_orders eo ON si.order_no = eo.order_no
    WHERE si.shipment_id = {{tbl_shipments.selectedRow.id || 0}}
),
FactoryMath AS (
    SELECT 
        *,
        -- Step 1: Calculate New Target
        GREATEST(0, target_qty - stock) AS new_target,
        
        -- Step 2: Calculate Current WIP (Work in Progress) sitting on the factory floor
        GREATEST(0, produced - (stock + sent)) AS current_prod_at_factory,
        GREATEST(0, folded - (stock + sent)) AS current_fold_at_factory
    FROM RawData
)
SELECT 
    order_no,
    
    -- THIS IS THE NEW MERGE LINE!
    CONCAT_WS(' ', item_size, color, logo, folding_type) AS item_description,
    
    cartons,
    target_qty AS required_pcs,
    
    -- THE PRINT CALCULATION (Applies the 8% rejection buffer to the new target)
    CASE 
        WHEN stock >= target_qty THEN 0
        ELSE GREATEST(0, CEIL((new_target * 1.08) - current_prod_at_factory))
    END AS qty_to_print,
    
    -- THE FOLD CALCULATION
    CASE 
        WHEN stock >= target_qty THEN 0
        ELSE GREATEST(0, new_target - current_fold_at_factory)
    END AS qty_to_fold,

    -- DYNAMIC FACTORY STATUS
    CASE 
        WHEN stock >= target_qty THEN '🟢 Ready to Load (From Stock)'
        WHEN GREATEST(0, new_target - current_fold_at_factory) <= 0 THEN '🟢 Ready to Load'
        WHEN GREATEST(0, CEIL((new_target * 1.08) - current_prod_at_factory)) <= 0 THEN '🟡 Needs Folding'
        ELSE '🔴 Needs Printing'
    END AS action_status

FROM FactoryMath
ORDER BY order_no ASC;
